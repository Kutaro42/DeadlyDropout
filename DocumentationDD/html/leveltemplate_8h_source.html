28// SYSTÈME DE TEMPLATE DE NIVEAU — VERSION 3 (matrices externalisées)
29//
30// Chaque niveau est défini par TROIS matrices de même dimension :
31//
32// entityMatrix (spawns et objets logiques) :
33// 0 = FLOOR (sol libre)
34// 2 = ZOMBIE_SPAWN
35// 3 = EXIT_DOOR
36// 4 = PLAYER_SPAWN
37// 5 = ZOMBIE_LAUNCHER_SPAWN
38// 6 = STANDARD_DOOR
39// 7 = BOSS_SPAWN
40// 8 = BOSS_DOOR
41//
42// wallMatrix :
43// 0 = pas de mur
44// N = index de tile dans le tileset de murs (1-based)
45//
46// floorMatrix :
47// 0 = sol uniforme (couleur de fond de scène, rien à dessiner)
48// N = index de tile dans le tileset de sol (1-based)
49// ═══════════════════════════════════════════════════════════════════
50
51// ─── LevelMatrix (rétrocompatibilité — non utilisé en runtime) ────
52// Conservé pour que le template makeDescriptor reste compilable si
53// quelqu'un souhaite encore définir un niveau directement en C++.
54template<int ROWS, int COLS>
55struct LevelMatrix {
56 static constexpr int rows = ROWS;
57 static constexpr int cols = COLS;
58 int data[ROWS][COLS];
59 int at(int r, int c) const { return data[r][c]; }
60};
